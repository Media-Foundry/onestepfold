import json,gzip,pathlib,numpy as np
r=pathlib.Path('reports/mini_folding_global_distance_evaluation_2026-09-30/final');cases=json.load(gzip.open(r/'extent_cases.json.gz'));report=json.load(gzip.open(r/'extent_report.json.gz'));lock=json.load(gzip.open(r/'lock.json.gz'))
assert len(cases)==455 and all(c['complete'] for c in cases)
lookup={}
for c in cases:
 for v in c['records']:lookup.setdefault((c['group_id'],v['model']),[]).append(v)
assert sum(map(len,lookup.values()))==5460 and all(len(v)==2 for v in lookup.values())
assert max(x['rmsd_replay_error'] for v in lookup.values() for x in v)==0
count=0;max_error=0
for co,s in report['summary'].items():
 for model,fields in s['models'].items():
  for metric,stats in fields.items():
   vals=[]
   for group in lock['cohorts'][co]:
    vv=[v[metric] if '.' not in metric else v['distance_bands'][metric.split('.')[0]][metric.split('.')[1]] for v in lookup[group,model]]
    if all(x is not None for x in vv):vals.append(float(np.mean(vv)))
   assert len(vals)==stats['proteins']
   if vals:
    error=abs(float(np.mean(vals))-stats['mean']);assert error<1e-12;max_error=max(max_error,error)
   count+=1
out=dict(complete=True,proteins=455,outputs=5460,rmsd_replay_max=0.,model_metric_cohort_means_checked=count,max_mean_error=max_error,scope='Independent aggregation of complete per-case diagnostic export; no new coordinates or inference; bootstrap not reimplemented.')
(r/'extent_local_acceptance.json').write_text(json.dumps(out,indent=2)+'\n');print(json.dumps(out))
