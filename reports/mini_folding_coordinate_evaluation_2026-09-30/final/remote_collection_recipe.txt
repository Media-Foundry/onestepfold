import json,hashlib,pathlib,gzip,subprocess
r=pathlib.Path('/data/user/shuang886/Folding/folding_coordinate_evaluation_v1_20260930')
def sha(p):
 h=hashlib.sha256()
 with open(p,'rb') as f:
  for b in iter(lambda:f.read(1048576),b''):h.update(b)
 return h.hexdigest()
p=json.loads((r/'final/provenance.json').read_text());checks={}
for path,expected in p['inputs'].items():
 actual=sha(path);assert actual==expected,path;checks[path]=actual
for name,key in [('report.md','report_sha256'),('paired.csv','paired_csv_sha256'),('global_structure.json','global_structure_sha256')]:
 assert sha(r/'final'/name)==p[key]
e=json.loads((r/'evaluation.json').read_text());assert e['complete'] and e['outputs']==4550 and e['proteins']==455 and e['counts']=={'coordinate_zero':910,'native':0} and e['probe_nfe']==32
assert e['metric_max_abs']<1e-12
rows=e['records'];keys=[(x['group_id'],x['seed'],x['model']) for x in rows];assert len(set(keys))==4550
# Keep all per-instance scalar and chemistry evidence; verbose per-edge arrays stay in hash-bound remote evaluation.
compact={k:v for k,v in e.items() if k not in ('records','paired','summary')}
compact['records']=[{k:v for k,v in row.items() if k!='connection_distribution'} for row in rows]
compact['source_evaluation_sha256']=sha(r/'evaluation.json')
with gzip.open(r/'final/scored_instances.json.gz','wt') as f:json.dump(compact,f,sort_keys=True)
with gzip.open(r/'final/cohorts.json.gz','wb') as f:f.write((r/'cohorts.json').read_bytes())
for name in ['lock.json','execution.json','scheduler_observation.json']:
 (r/'final'/name).write_bytes((r/name).read_bytes())
(r/'final/ablation_training_audit.json').write_bytes(pathlib.Path('/data/user/shuang886/Folding/folding_coordinate_ablation_v1_20260930/ablation_training_audit.json').read_bytes())
jobs='662470,662471,662475,662476,662477,662478,662479,662480,662481,662482,662483,662484,662485'
s=subprocess.check_output(['sacct','-X','-n','-j',jobs,'--format=JobID,State,ExitCode,Elapsed','-P'],text=True)
a=[line.split('|') for line in s.strip().splitlines()];assert len(a)==13 and all(x[1:3]==['COMPLETED','0:0'] for x in a)
(r/'final/scheduler.txt').write_text(s)
manifest={'complete':True,'source_hashes_rechecked':checks,'files':{x.name:sha(x) for x in (r/'final').iterdir() if x.is_file() and x.name!='collection.json'},'counts_verified':{'proteins':455,'outputs':4550,'new_prediction_nfes':910,'engineering_nfes':32},'scope':'hashes, completed jobs and scored-output identities; no new inference; full per-edge distributions retained remotely'}
(r/'final/collection.json').write_text(json.dumps(manifest,indent=2)+'\n')
print(json.dumps({'complete':True,'files':len(manifest['files']),'metric_max_abs':e['metric_max_abs']}))
