import json,gzip,hashlib,pathlib,numpy as np
r=pathlib.Path('reports/mini_folding_structure_extent_2026-09-30');accept=json.loads((r/'acceptance.json').read_text())
for n,h in accept['files'].items(): assert hashlib.sha256((r/n).read_bytes()).hexdigest()==h,n
cases=json.load(gzip.open(r/'cases.json.gz'));report=json.loads((r/'report.json').read_text());assert len(cases)==455 and all(c['complete'] for c in cases)
lookup={(c['group_id'],v['model'],v['seed']):v for c in cases for v in c['records']};assert len(lookup)==4550
assert max(x['rmsd_replay_error'] for x in lookup.values())==0
cohorts={co:[x['group_id'] for x in report['paired'] if x['cohort']==co] for co in report['summary']}
count=0
for co,s in report['summary'].items():
 for model,fields in s['models'].items():
  for metric,stats in fields.items():
   vals=[]
   for group in cohorts[co]:
    rr=[v for (g,m,seed),v in lookup.items() if g==group and m==model];assert len(rr)==2
    vv=[v[metric] if '.' not in metric else v['distance_bands'][metric.split('.')[0]][metric.split('.')[1]] for v in rr]
    if all(v is not None for v in vv):vals.append(float(np.mean(vv)))
   assert len(vals)==stats['proteins']
   if vals:assert abs(float(np.mean(vals))-stats['mean'])<1e-12
   count+=1
out=dict(complete=True,archive_files_checked=len(accept['files']),proteins=455,outputs=4550,rmsd_replay_max=0.,model_metric_cohort_means_checked=count,scope='independent aggregation from per-case archive; coordinate RMSD replay performed remotely; no new inference')
(r/'local_acceptance.json').write_text(json.dumps(out,indent=2)+'\n');print(json.dumps(out))
