from pathlib import Path
import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
root=Path('reports/mini_reference_editor_intervention_2026-10-07')
d=pd.read_csv(root/'summary.csv')
runs=['n3_workspace_272001','n3_workspace_272003','n15_workspace_272001','n15_workspace_272003']
labels=['3 ref / seed 1','3 ref / seed 2','15 ref / seed 1','15 ref / seed 2']
fig,axes=plt.subplots(1,2,figsize=(12,4.3),layout='constrained')
for ax,metric,title in zip(axes,['spearman','regret'],['19-AA ranking (higher is better)','Old-select / new-Exact regret (lower is better)']):
 for ai,(arm,color) in enumerate(zip(['correct','common','shuffled'],['#24618e','#dd9c36','#648b76'])):
  vals=[float(d[(d.run==r)&(d.arm==arm)][metric].iloc[0]) for r in runs]
  ax.bar(np.arange(4)+(ai-1)*.24,vals,width=.23,label=arm,color=color)
 baseline=float(d[d.arm=='reference_only'][metric].iloc[0])
 ax.axhline(baseline,color='#a53c45',ls='--',label='reference-only')
 ax.axhline(0,color='#777',lw=.6)
 ax.set_xticks(range(4),labels,rotation=12);ax.set_title(title);ax.grid(axis='y',alpha=.15);ax.set_axisbelow(True)
axes[0].legend(frameon=False,fontsize=9,ncol=2)
fig.suptitle('Fixed terminal workspace interventions | 9 reused development proteins / 18 sites',fontsize=12)
fig.savefig(root/'intervention_comparison.png',dpi=190)
fig.savefig(root/'intervention_comparison.pdf')
(root/'plot_source.py.txt').write_bytes(Path(__file__).read_bytes())
